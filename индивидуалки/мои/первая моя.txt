#include <iostream>
#include <fstream>
#include <vector>
#include <clocale>

using namespace std;

int main() {
    setlocale(LC_ALL, "Russian");

    // Открываем файл
    ifstream input("matrix.txt");

    if (!input) {
        cerr << "Не удалось открыть файл matrix.txt" << endl;
        return 1;
    }

    // Считываем количество уравнений и неизвестных
    int rows, cols;
    input >> rows >> cols;

    // Создаем матрицу коэффициентов и столбец свободных членов
    vector<vector<double>> matrix(rows, vector<double>(cols));
    vector<double> rhs(rows);

    // Считываем данные из файла
    for (int i = 0; i < rows; ++i) {
        for (int j = 0; j < cols; ++j) {
            input >> matrix[i][j];
        }

        input >> rhs[i];
    }

    input.close();

    // Вывод расширенной матрицы
    cout << "Расширенная матрица:" << endl;

    for (int i = 0; i < rows; ++i) {
        for (int j = 0; j < cols; ++j) {
            cout << matrix[i][j] << "\t";
        }

        cout << "| " << rhs[i] << endl;
    }

    cout << endl;

    // Прямой ход метода Гаусса
    for (int k = 0; k < cols - 1; ++k) {

        if (matrix[k][k] == 0.0) {
            cerr << "Ведущий элемент равен нулю" << endl;
            return 1;
        }

        for (int i = k + 1; i < rows; ++i) {

            double factor = matrix[i][k] / matrix[k][k];

            // Преобразуем свободный член
            rhs[i] -= factor * rhs[k];

            // Преобразуем коэффициенты
            for (int j = k; j < cols; ++j) {
                matrix[i][j] -= factor * matrix[k][j];
            }
        }
    }

   

    // Обратный ход
    vector<double> solution(cols);

    for (int i = cols - 1; i >= 0; --i) {

        double value = rhs[i];

        // Вычитаем уже найденные неизвестные
        for (int j = i + 1; j < cols; ++j) {
            value -= matrix[i][j] * solution[j];
        }

        if (matrix[i][i] == 0.0) {
            cerr << "Деление на ноль при нахождении решения" << endl;
            return 1;
        }

        solution[i] = value / matrix[i][i];
    }

    // Вывод решения
    cout << "Решение системы:" << endl;

    for (int i = 0; i < cols; ++i) {
        cout << "x" << i + 1 << " = " << solution[i] << endl;
    }

    return 0;
}
